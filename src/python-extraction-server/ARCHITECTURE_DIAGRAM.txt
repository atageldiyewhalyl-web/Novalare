                        TEMPLATE PRIORITY SYSTEM

    │  1. User-Learned Templates (Highest Priority)   │
    │     /templates/user_learned/*.json              │
    │     • Organization-specific                     │
    │     • Override built-in templates               │
    │     • Tier 3 auto-saves here                    │
    └─────────────────────────────────────────────────┘
                            ↓ (if not found)
    ┌─────────────────────────────────────────────────┐
    │  2. Built-In Templates                          │
    │     /templates/built_in/*.json                  │
    │     • Official bank templates                   │
    │     • Maintained by Novalare                    │
    │     • Community contributions                   │
    └─────────────────────────────────────────────────┘
                            ↓ (if not found)
    ┌─────────────────────────────────────────────────┐
    │  3. AI Discovery (Tier 2)                       │
    │     • GPT-4 Vision analyzes layout              │
    │     • Creates template on-the-fly               │
    │     • Saves to user_learned/ for future         │
    └─────────────────────────────────────────────────┘
                            ↓ (if AI fails)
    ┌─────────────────────────────────────────────────┐
    │  4. Manual Mapping (Tier 3)                     │
    │     • User clicks sample transactions           │

    │     • Saves to user_learned/                    │
    └─────────────────────────────────────────────────┘

                          TEMPLATE JSON FORMAT

{
  "bank_key": "capital_one",              ← Unique identifier
  "bank_name": "Capital One",             ← Display name
  "version": "1.1",                       ← Template version
  "created_by": "system",                 ← system | user | ai
  "last_updated": "2024-12-22",           ← Last modified
  
  "statement_model": "running_balance",   ← running_balance | soll_haben
  "currency": "USD",                      ← USD | EUR | GBP
  
  "columns": {                            ← X-coordinates (pdfplumber)
    "date": {
      "x_min": 70,                        ← Left edge
      "x_max": 105,                       ← Right edge
      "note": "Month name: 'Apr 1'"       ← Developer notes
    },
    "description": {
      "x_min": 125,
      "x_max": 500
    },
    "amount": {
      "x_min": 510,
      "x_max": 580
    },
    "balance": {
      "x_min": 640,
      "x_max": 710
    }
  },
  
  "transaction_start_markers": [          ← Header text to find
    "DATE",
    "DESCRIPTION"
  ],
  
  "date_format": "MMM D",                 ← MM/DD | DD.MM.YYYY | MMM D
  "has_balance_column": true,             ← true | false
  "multiline_descriptions": false,        ← true | false
  
  "detection_keywords": [                 ← Auto-detect bank
    "capital one",
    "capitalone"
  ]
}
